import hashlib
import json
import pathlib
import zipfile

ROOT = pathlib.Path('${VALIDATION_ROOT}')
OUT = ROOT/'baseline-cpu-comparison'
HEAD = ROOT/'cpu-validation'

def read(path):
    return json.loads(path.read_text())

def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def archive(path):
    with zipfile.ZipFile(path) as file:
        return {name:hashlib.sha256(file.read(name)).hexdigest() for name in file.namelist()}

def semantic(state):
    observers = {}
    fields = ['enabled','fake_quant_enabled','dtype','qscheme','channel_axis','_qparams_locked']
    for name, obs in state['observers'].items():
        item = {key:obs.get(key) for key in fields}
        for key in ['_cached_scale','_cached_zp','min_val','max_val']:
            if key in obs.get('buffers',{}):
                item[key] = obs['buffers'][key]
            elif key in obs:
                item[key] = obs[key]
        observers[name] = item
    return {'weights':state['weights'],'modes':state['modes'],'observers':observers}

def canonical_observers(state):
    observers=semantic(state)['observers']
    for name, fields in observers.items():
        raw=state['observers'][name]
        if 'snapshot' in raw:
            snapshot=raw['snapshot']
            fields['_qparams_locked']=repr(snapshot['locked']) if snapshot else 'False'
    return observers

def differences(left,right):
    result=[]
    for name in sorted(set(left)|set(right)):
        if name not in left or name not in right:
            result.append({'observer':name,'field':'observer_presence','base':name in left,'head':name in right})
            continue
        for field in sorted(set(left[name])|set(right[name])):
            if left[name].get(field)!=right[name].get(field):
                result.append({'observer':name,'field':field,'base':left[name].get(field),'head':right[name].get(field)})
    return result

Qwen additionally removes two unused vision scale observers, leaves four fused text scale/raw-logit observers inactive, and leaves unused DeepStack layer1 without fabricated qparams; these exact differences are preserved.',

                'same_probes_and_torch_parity':bmeta['probes']==hmeta['probes'],
                'same_circle_contract':bmeta['contract']==hmeta['contract'],
                'base_circle_sha256':digest(bphase/'artifacts'/artifact),
                'head_circle_sha256':digest(hphase/'artifacts'/artifact),
                'array_sha256_base':barrays,'array_sha256_head':harrays,

        data['phases'][phase]={'base_state_preserved':bsnap['state_preserved'],'head_state_preserved':hsnap['state_preserved'],'stages':comparisons}
    arrays_equal=all(s.get('same_raw_input_torch_circle_npy_hashes',False) for p in data['phases'].values() for s in p['stages'].values())
    data['same_all_numerical_inputs_and_outputs']=arrays_equal
    data['classification']='기존 수치 parity 실패 (최신 base에서 동일 weight·활성 frozen grid·입력·Torch/Circle 출력과 같은 strict parity 실패 재현; 비활성/schema metadata 차이는 별도 보존)' if all([data['base_status']=='FAIL',data['head_status']=='FAIL',data['same_parameter_hashes'],data['same_quant_modes'],data['same_active_frozen_observer_grids_policy_lock_ranges'],data['same_failure_stages'],arrays_equal]) else '미해결: detailed differences 확인 필요'
